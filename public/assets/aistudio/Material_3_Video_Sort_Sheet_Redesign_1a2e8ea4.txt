// UPDATED SMART RESOLUTION LOGIC based on user feedback
const getSmartResolutionTier = (width, height) => {
  const longEdge = Math.max(width, height);
  if (longEdge >= 7000) return '8K';
  if (longEdge >= 3200) return '4K';
  if (longEdge >= 2200) return '1440p'; // Changed from 2K to 1440p
  if (longEdge >= 1500) return '1080p'; // Lowered threshold from 1600 to 1500
  if (longEdge >= 1100) return '720p';
  return 'SD';
};

const formatCompactDate = (dateStr) => {
  if (!dateStr) return '';
  return dateStr
    .replace('January', 'Jan')
    .replace('February', 'Feb')
    .replace('March', 'Mar')
    .replace('April', 'Apr')
    .replace('June', 'Jun')
    .replace('July', 'Jul')
    .replace('August', 'Aug')
    .replace('September', 'Sept')
    .replace('October', 'Oct')
    .replace('November', 'Nov')
    .replace('December', 'Dec');
};

export default function Material3VideoPlayer() {
  const [videos, setVideos] = useState(INITIAL_VIDEOS);

  // Core View & Sort states
  const [smartGroups, setSmartGroups] = useState(false);
  const [viewMode, setViewMode] = useState('videos'); 
  const [layoutMode, setLayoutMode] = useState('list'); 
  const [sortBy, setSortBy] = useState('date'); 
  const [sortDir, setSortDir] = useState('desc'); 

  // Material You Theme & Display states
  const [colorTheme, setColorTheme] = useState('indigo');
  const [amoledBlack, setAmoledBlack] = useState(true); 
  const [showResolutionTier, setShowResolutionTier] = useState(true);

  // Navigation & Search states
  const [bottomTab, setBottomTab] = useState('videos');
  const [searchOpen, setSearchOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');

  // Overlays & Contextual Action states
  const [selectedIds, setSelectedIds] = useState([]);
  const [activeMenuVideo, setActiveMenuVideo] = useState(null);
  const [detailsVideo, setDetailsVideo] = useState(null);
  const [renameTarget, setRenameTarget] = useState(null);
  const [renameValue, setRenameValue] = useState('');
  const [deleteConfirmOpen, setDeleteConfirmOpen] = useState(false);
  const [playlistTargetIds, setPlaylistTargetIds] = useState([]);
  const [sortSheetOpen, setSortSheetOpen] = useState(false);
  const [themeSheetOpen, setThemeSheetOpen] = useState(false);
  const [toastMsg, setToastMsg] = useState(null);

  const palette = M3_PALETTES[colorTheme] || M3_PALETTES.indigo;
  const isSelectionMode = selectedIds.length > 0;

  // Dynamic M3 surface tokens
  const outerChromeBg = amoledBlack ? '#000000' : palette.surfaceDim;
  const sheetBg = amoledBlack ? '#000000' : palette.surface;
  const rowCardBg = amoledBlack ? '#000000' : palette.surfaceContainerLow;

  const triggerToast = (msg) => {
    setToastMsg(msg);
    setTimeout(() => setToastMsg(null), 2400);
  };

  const clearAllOverlays = () => {
    setActiveMenuVideo(null);
    setDetailsVideo(null);
    setRenameTarget(null);
    setDeleteConfirmOpen(false);
    setPlaylistTargetIds([]);
    setSortSheetOpen(false);
    setThemeSheetOpen(false);
  };

  const filteredAndSortedVideos = useMemo(() => {
    let list = [...videos];
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      list = list.filter(
        (v) => v.title.toLowerCase().includes(q) || v.folder.toLowerCase().includes(q)
      );
    }
    list.sort((a, b) => {
      let cmp = 0;
      if (sortBy === 'name') cmp = a.title.localeCompare(b.title);
      else if (sortBy === 'date') cmp = a.timestamp - b.timestamp;
      else if (sortBy === 'duration') cmp = a.durationSec - b.durationSec;
      else if (sortBy === 'size') cmp = a.sizeMB - b.sizeMB;
      return sortDir === 'asc' ? cmp : -cmp;
    });
    return list;
  }, [videos, searchQuery, sortBy, sortDir]);

  const smartSeriesData = useMemo(() => {
    const groups = {};
    const standalone = [];
    filteredAndSortedVideos.forEach((v) => {
      if (v.series) {
        if (!groups[v.series]) groups[v.series] = [];
        groups[v.series].push(v);
      } else {
        standalone.push(v);
      }
    });
    Object.keys(groups).forEach((key) => {
      groups[key].sort((a, b) => (a.episode || 0) - (b.episode || 0));
    });
    return { groups, standalone };
  }, [filteredAndSortedVideos]);

  const handleSortOptionClick = (key) => {
    if (sortBy === key) {
      setSortDir((prev) => (prev === 'asc' ? 'desc' : 'asc'));
    } else {
      setSortBy(key);
      setSortDir(key === 'name' ? 'asc' : 'desc');
    }
  };

  const toggleSelectId = (id) => {
    setSelectedIds((prev) =>
      prev.includes(id) ? prev.filter((item) => item !== id) : [...prev, id]
    );
  };

  const confirmDelete = () => {
    let count = 0;
    if (isSelectionMode) {
      count = selectedIds.length;
      setVideos((prev) => prev.filter((v) => !selectedIds.includes(v.id)));
      setSelectedIds([]);
    } else if (activeMenuVideo) {
      count = 1;
      setVideos((prev) => prev.filter((v) => v.id !== activeMenuVideo.id));
    }
    clearAllOverlays();
    triggerToast(`Deleted ${count} video${count > 1 ? 's' : ''}`);
  };

  const confirmRename = (e) => {
    e.preventDefault();
    if (!renameTarget || !renameValue.trim()) return;
    setVideos((prev) =>
      prev.map((v) => (v.id === renameTarget.id ? { ...v, title: renameValue.trim() } : v))
    );
    clearAllOverlays();
    triggerToast('File renamed');
  };

  const renderVideoItem = (video, index, totalCount, badgeOverride = null) => {
    const isSelected = selectedIds.includes(video.id);
    const tier = getSmartResolutionTier(video.width, video.height);
    const shortDate = formatCompactDate(video.dateStr);

    // M3 Expressive Segmented Containment Radii
    // When selected (even in AMOLED mode), always use a distinct 16px rounded tray so adjacent selected items never merge
    const isFirst = index === 0;
    const isLast = index === totalCount - 1;
    let itemRadiusClass = 'rounded-[16px]';
    if (!isSelected) {
      if (amoledBlack) {
        itemRadiusClass = 'rounded-[12px]';
      } else if (totalCount === 1) {
        itemRadiusClass = 'rounded-[16px]';
      } else if (isFirst) {
        itemRadiusClass = 'rounded-t-[16px] rounded-b-[4px]';
      } else if (isLast) {
        itemRadiusClass = 'rounded-t-[4px] rounded-b-[16px]';
      } else {
        itemRadiusClass = 'rounded-[4px]';
      }
    }

    if (layoutMode === 'grid') {
      return (
        <div
          key={video.id}
          onClick={() => (isSelectionMode ? toggleSelectId(video.id) : triggerToast(`Playing "${video.title}"`))}
          onContextMenu={(e) => {
            e.preventDefault();
            clearAllOverlays();
            toggleSelectId(video.id);
          }}
          style={{
            backgroundColor: isSelected ? palette.secondaryContainer : amoledBlack ? '#000000' : palette.surfaceContainerLow
          }}
          className="group relative flex flex-col gap-2 p-2 rounded-[16px] transition-all duration-200 active:scale-[0.98] cursor-pointer"
        >
          <div
            className="relative w-full aspect-video rounded-[12px] overflow-hidden"
            style={{ background: video.thumbStyle.bg }}
          >
            <div className="absolute inset-0 flex flex-col items-center justify-center p-2 text-center pointer-events-none">
              <span className="text-[10px] font-bold tracking-widest uppercase text-white/75 drop-shadow-md">
                {video.thumbStyle.tag}
              </span>
            </div>
            
            <div className="absolute inset-x-0 bottom-0 h-8 bg-gradient-to-t from-black/80 via-black/20 to-transparent pointer-events-none" />
            <span className="absolute bottom-1.5 right-2 text-[11px] font-medium text-white tracking-wide tabular-nums drop-shadow-md">
              {video.durationLabel}
            </span>
            {video.progress > 0 && (
              <div className="absolute inset-x-0 bottom-0 h-[3px] bg-white/20">
                <div
                  style={{ width: `${video.progress * 100}%`, backgroundColor: palette.primary }}
                  className="h-full"
                />
              </div>
            )}
          </div>

          <div className="flex items-start justify-between gap-1 px-0.5">
            <div className="min-w-0 flex-1">
              <p
                style={{ color: palette.onSurface }}
                className="text-[13px] font-medium leading-[1.3] line-clamp-2 break-words"
              >
                {video.title}
              </p>
              <div
                style={{ color: palette.onSurfaceVariant }}
                className="flex items-center gap-1.5 mt-1 text-[11.5px] whitespace-nowrap overflow-hidden"
              >
                {showResolutionTier && tier !== 'SD' && (
                  <span className="font-semibold tracking-tight flex-shrink-0">{tier}</span>
                )}
                <span className="truncate">{shortDate}</span>
              </div>
            </div>
            {isSelectionMode ? (
              <div
                style={{
                  backgroundColor: isSelected ? palette.primary : 'transparent',
                  borderColor: isSelected ? palette.primary : palette.outline,

                        }`}
                      >
                        <div className="flex items-center gap-2.5 min-w-0">
                          <Icon className="w-[18px] h-[18px] flex-shrink-0" />
                          <span className="text-[14.5px] font-medium truncate">{cfg.label}</span>
                        </div>
                        {active && (
                          <span
                            style={{ backgroundColor: palette.surfaceContainerLowest, color: palette.primary }}
                            className="h-7 px-2.5 rounded-full flex items-center gap-1.5 text-[11.5px] font-bold whitespace-nowrap flex-shrink-0"
                          >
                            <span>{dirBadge}</span>
                            {sortDir === 'asc' ? <ArrowUp className="w-3.5 h-3.5 stroke-[2.5]" /> : <ArrowDown className="w-3.5 h-3.5 stroke-[2.5]" />}
                          </span>
                        )}
                      </button>
                    );
                  })}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* RESTORED ColorOS-Style Details Sheet */}
        {detailsVideo && (
          <div className="absolute inset-0 z-50 flex flex-col justify-end">
            <div onClick={clearAllOverlays} className="absolute inset-0 bg-black/60" />
            <div
              style={{ backgroundColor: palette.surfaceContainerLow }}
              className="relative z-10 rounded-t-[28px] shadow-2xl animate-[m3SheetUp_0.24s_cubic-bezier(0.2,0,0,1)] pb-8"
            >
              {/* Header */}
              <div className="flex items-center justify-between px-5 pt-4 pb-2">
                <button
                  type="button"
                  onClick={clearAllOverlays}
                  style={{ color: palette.primary }}
                  className="text-[15px] font-medium active:scale-95 transition-transform"
                >
                  Close
                </button>
                <h2 style={{ color: palette.onSurface }} className="text-[17px] font-medium absolute left-1/2 -translate-x-1/2">
                  Details
                </h2>
                <div className="w-10" />
              </div>

              {/* Minimal Rounded Card */}
              <div className="px-4 pt-2">
                <div style={{ backgroundColor: palette.surfaceContainer }} className="rounded-[20px] px-5 py-2">
                  {[
                    { label: 'Name', value: `${detailsVideo.title}.${detailsVideo.ext}` },
                    { label: 'Format', value: detailsVideo.ext },
                    { label: 'Size', value: `${detailsVideo.sizeMB} MB` },
                    { label: 'Resolution', value: `${detailsVideo.width} × ${detailsVideo.height} pixels [${getSmartResolutionTier(detailsVideo.width, detailsVideo.height)}]` },
                    { label: 'Orientation', value: detailsVideo.height > detailsVideo.width ? 'Vertical (9:16)' : 'Landscape (16:9)' },
                    { label: 'Length', value: detailsVideo.durationLabel },
                    { label: 'Path', value: detailsVideo.path, copyable: true }
                  ].map((row, idx, arr) => (
                    <div
                      key={row.label}
                      className={`flex flex-col py-3.5 ${idx !== arr.length - 1 ? 'border-b' : ''}`}
                      style={{ borderColor: palette.surfaceContainerHighest }}
                    >
                      <span style={{ color: palette.onSurfaceVariant }} className="text-[13px] mb-0.5">
                        {row.label}
                      </span>
                      <div className="flex items-start justify-between gap-4">
                        <span style={{ color: palette.onSurface }} className="text-[15px] font-medium break-all leading-snug">
                          {row.value}
                        </span>
                        {row.copyable && (
                          <button
                            type="button"
                            onClick={() => triggerToast('Path copied')}
                            style={{ backgroundColor: palette.secondaryContainer, color: palette.onSecondaryContainer }}
                            className="px-3 py-1.5 rounded-full text-[12px] font-medium flex-shrink-0"
                          >
                            Copy
                          </button>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* RESTORED ColorOS-Style Rename Dialog */}
        {renameTarget && (
          <div className="absolute inset-0 z-50 flex items-center justify-center px-6">
            <div onClick={clearAllOverlays} className="absolute inset-0 bg-black/60" />
            <form
              onSubmit={confirmRename}
              style={{ backgroundColor: palette.surfaceContainerHigh }}
              className="relative z-10 w-full rounded-[24px] p-6 shadow-2xl animate-[m3Pop_0.18s_ease-out]"
            >
              <h3 style={{ color: palette.onSurface }} className="text-[20px] font-medium mb-4">
                Rename video
              </h3>
              <div
                style={{ backgroundColor: palette.surfaceContainer, borderColor: palette.primary }}
                className="rounded-[12px] border-b-2 px-3 py-2 mb-6"
              >
                <label style={{ color: palette.primary }} className="block text-[11px] font-medium mb-1">
                  File name (.{renameTarget.ext})
                </label>
                <input
                  type="text"
                  value={renameValue}
                  onChange={(e) => setRenameValue(e.target.value)}
                  autoFocus
                  style={{ color: palette.onSurface }}
                  className="w-full bg-transparent text-[16px] focus:outline-none"
                />
              </div>
              <div className="flex items-center justify-end gap-2">
                <button
                  type="button"
                  onClick={clearAllOverlays}
                  style={{ color: palette.primary }}
                  className="px-5 py-2.5 rounded-full text-[14px] font-medium"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  style={{ backgroundColor: palette.primary, color: palette.onPrimary }}
                  className="px-6 py-2.5 rounded-full text-[14px] font-medium active:scale-95"
                >
                  Save
                </button>
              </div>
            </form>
          </div>
        )}

        {/* RESTORED ColorOS-Style Delete Dialog */}
        {deleteConfirmOpen && (
          <div className="absolute inset-0 z-50 flex items-center justify-center px-6">
            <div onClick={clearAllOverlays} className="absolute inset-0 bg-black/60" />
            <div
              style={{ backgroundColor: palette.surfaceContainerHigh }}
              className="relative z-10 w-full rounded-[24px] p-6 shadow-2xl animate-[m3Pop_0.18s_ease-out]"
            >
              <h3 style={{ color: palette.onSurface }} className="text-[20px] font-medium text-center mb-6">
                {isSelectionMode ? `Delete these ${selectedIds.length} videos?` : 'Delete this video?'}
              </h3>
              <div className="flex items-center gap-3">
                <button
                  type="button"
                  onClick={clearAllOverlays}
                  style={{ backgroundColor: palette.surfaceContainerHighest, color: palette.onSurface }}
                  className="flex-1 py-3 rounded-full text-[15px] font-medium active:scale-95"
                >
                  Cancel
                </button>
                <button
                  type="button"
                  onClick={confirmDelete}
                  style={{ backgroundColor: palette.error, color: '#490002' }}
                  className="flex-1 py-3 rounded-full text-[15px] font-bold active:scale-95"
                >
                  Delete
                </button>
              </div>
            </div>
          </div>
        )}

        {/* Playlist & Theme Sheets */}
        {playlistTargetIds.length > 0 && (
          <div className="absolute inset-0 z-50 flex items-center justify-center px-6">
            <div onClick={clearAllOverlays} className="absolute inset-0 bg-black/60" />
            <div style={{ backgroundColor: palette.surfaceContainerHigh }} className="relative z-10 w-full rounded-[24px] p-6 shadow-2xl animate-[m3Pop_0.18s_ease-out]">
              <h3 style={{ color: palette.onSurface }} className="text-[20px] font-medium mb-4">Add to playlist</h3>
              <div className="space-y-[2px] mb-5">
                {['Watch Later', 'Favorites', 'Cinema 4K', 'Study & Tutorials'].map((plName, idx, arr) => (
                  <button
                    key={plName}
                    type="button"
                    onClick={() => { clearAllOverlays(); setSelectedIds([]); triggerToast(`Added to "${plName}"`); }}
                    style={{ backgroundColor: palette.surfaceContainer, color: palette.onSurface }}
                    className={`w-full px-4 py-3.5 text-left text-[15px] font-medium flex items-center justify-between hover:opacity-90 ${
                      idx === 0 ? 'rounded-t-[16px] rounded-b-[4px]' : idx === arr.length - 1 ? 'rounded-t-[4px] rounded-b-[16px]' : 'rounded-[4px]'
                    }`}
                  >
                    <span>{plName}</span>
                    <FolderPlus style={{ color: palette.primary }} className="w-[18px] h-[18px]" />
                  </button>
                ))}
              </div>
              <div className="flex justify-end">
                <button type="button" onClick={clearAllOverlays} style={{ color: palette.primary }} className="px-5 py-2.5 rounded-full text-[14px] font-medium">Cancel</button>
              </div>
            </div>
          </div>
        )}

        {themeSheetOpen && (
          <div className="absolute inset-0 z-50 flex flex-col justify-end">
            <div onClick={clearAllOverlays} className="absolute inset-0 bg-black/60" />
            <div style={{ backgroundColor: palette.surfaceContainerLow }} className="relative z-10 rounded-t-[28px] px-5 pt-3 pb-8 shadow-2xl animate-[m3SheetUp_0.24s_cubic-bezier(0.2,0,0,1)]">
              <div className="w-8 h-1 rounded-full bg-[#8E9099]/50 mx-auto mb-5" />
              
              <div className="mb-6">
                <p style={{ color: palette.onSurfaceVariant }} className="text-[13px] font-medium mb-3">Dynamic Color Seed</p>
                <div className="flex overflow-x-auto gap-2.5 pb-2 scrollbar-hide">
                  {Object.values(M3_PALETTES).map((p) => {
                    const active = colorTheme === p.id;
                    return (
                      <button
                        key={p.id}
                        onClick={() => setColorTheme(p.id)}
                        style={{ backgroundColor: active ? p.secondaryContainer : palette.surfaceContainer, borderColor: active ? p.primary : 'transparent' }}
                        className="px-3 py-2 rounded-[16px] border-2 flex flex-col items-center gap-2 flex-shrink-0 w-[72px]"
                      >
                        <div style={{ backgroundColor: p.primary }} className="w-8 h-8 rounded-full flex items-center justify-center">
                          {active && <Check style={{ color: p.onPrimary }} className="w-4 h-4 stroke-[3]" />}
                        </div>
                        <span style={{ color: palette.onSurface }} className="text-[10px] font-medium truncate w-full text-center">
                          {p.name.split(' ')[0]}
                        </span>
                      </button>
                    );
                  })}
                </div>
              </div>

              <div className="space-y-[2px]">
                <div
                  onClick={() => setAmoledBlack(!amoledBlack)}
                  style={{ backgroundColor: palette.surfaceContainer }}
                  className="p-4 rounded-t-[20px] rounded-b-[4px] flex items-center justify-between cursor-pointer"
                >
                  <div>
                    <p style={{ color: palette.onSurface }} className="text-[15px] font-medium">AMOLED Pure Black</p>
                    <p style={{ color: palette.onSurfaceVariant }} className="text-[12px] mt-0.5">Remove gray card containers</p>
                  </div>
                  <div style={{ backgroundColor: amoledBlack ? palette.primary : palette.surfaceContainerHighest }} className="w-[52px] h-[32px] rounded-full p-1 flex items-center transition-colors">
                    <div style={{ backgroundColor: amoledBlack ? palette.onPrimary : palette.outline }} className={`w-6 h-6 rounded-full transition-transform ${amoledBlack ? 'translate-x-5' : 'translate-x-0'}`} />
                  </div>
                </div>
                <div
                  onClick={() => setShowResolutionTier(!showResolutionTier)}
                  style={{ backgroundColor: palette.surfaceContainer }}
                  className="p-4 rounded-t-[4px] rounded-b-[20px] flex items-center justify-between cursor-pointer"
                >
                  <div>
                    <p style={{ color: palette.onSurface }} className="text-[15px] font-medium">Show Resolution</p>
                    <p style={{ color: palette.onSurfaceVariant }} className="text-[12px] mt-0.5">Show 4K, 1440p, 1080p text</p>
                  </div>
                  <div style={{ backgroundColor: showResolutionTier ? palette.primary : palette.surfaceContainerHighest }} className="w-[52px] h-[32px] rounded-full p-1 flex items-center transition-colors">
                    <div style={{ backgroundColor: showResolutionTier ? palette.onPrimary : palette.outline }} className={`w-6 h-6 rounded-full transition-transform ${showResolutionTier ? 'translate-x-5' : 'translate-x-0'}`} />
                  </div>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* Toast */}
        {toastMsg && (
          <div className="absolute inset-x-4 bottom-28 z-50 flex justify-center pointer-events-none animate-[m3Pop_0.16s_ease-out]">
            <div style={{ backgroundColor: palette.onSurface, color: palette.surface }} className="px-4 py-2.5 rounded-[12px] text-[14px] font-medium shadow-xl">
              {toastMsg}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
